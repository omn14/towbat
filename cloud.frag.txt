#version 150

// Inputs from vertex shader
in vec2 texcoord;
in vec4 worldPos;

// Uniforms
uniform float customTime;
uniform vec4 cloudColor;
uniform vec4 skyColor;
uniform float cloudCoverage;  // 0.0 = no clouds, 1.0 = full coverage

// Output
out vec4 fragColor;

// Simplex noise function (approximation)
vec3 mod289(vec3 x) {
    return x - floor(x * (1.0 / 289.0)) * 289.0;
}

vec2 mod289(vec2 x) {
    return x - floor(x * (1.0 / 289.0)) * 289.0;
}

vec3 permute(vec3 x) {
    return mod289(((x * 34.0) + 1.0) * x);
}

float snoise(vec2 v) {
    const vec4 C = vec4(0.211324865405187,  // (3.0-sqrt(3.0))/6.0
                        0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
                        -0.577350269189626, // -1.0 + 2.0 * C.x
                        0.024390243902439); // 1.0 / 41.0
    
    vec2 i  = floor(v + dot(v, C.yy));
    vec2 x0 = v - i + dot(i, C.xx);
    
    vec2 i1;
    i1 = (x0.x > x0.y) ? vec2(1.0, 0.0) : vec2(0.0, 1.0);
    vec4 x12 = x0.xyxy + C.xxzz;
    x12.xy -= i1;
    
    i = mod289(i);
    vec3 p = permute(permute(i.y + vec3(0.0, i1.y, 1.0))
                    + i.x + vec3(0.0, i1.x, 1.0));
    
    vec3 m = max(0.5 - vec3(dot(x0, x0), dot(x12.xy, x12.xy), dot(x12.zw, x12.zw)), 0.0);
    m = m * m;
    m = m * m;
    
    vec3 x = 2.0 * fract(p * C.www) - 1.0;
    vec3 h = abs(x) - 0.5;
    vec3 ox = floor(x + 0.5);
    vec3 a0 = x - ox;
    
    m *= 1.79284291400159 - 0.85373472095314 * (a0 * a0 + h * h);
    
    vec3 g;
    g.x = a0.x * x0.x + h.x * x0.y;
    g.yz = a0.yz * x12.xz + h.yz * x12.yw;
    return 130.0 * dot(m, g);
}

// Fractal Brownian Motion for cloud detail
float fbm(vec2 p) {
    float value = 0.0;
    float amplitude = 0.5;
    float frequency = 1.0;
    
    // 5 octaves of noise
    for(int i = 0; i < 5; i++) {
        value += amplitude * snoise(p * frequency);
        frequency *= 2.0;
        amplitude *= 0.5;
    }
    
    return value;
}

void main() {
    // Scroll clouds over time
    float time = customTime * 0.05;
    vec2 uv = texcoord * 3.0;
    
    // Animate clouds
    vec2 q = vec2(fbm(uv + vec2(time, time * 0.5)),
                  fbm(uv + vec2(time * 0.8, -time * 0.3)));
    
    // Create cloud pattern with multiple layers
    float cloudPattern = fbm(uv + q * 2.0 + vec2(time * 0.1, 0.0));
    
    // Add more detail
    cloudPattern += fbm(uv * 2.0 + q + vec2(-time * 0.15, time * 0.1)) * 0.5;
    
    // Normalize and adjust contrast
    cloudPattern = (cloudPattern + 1.0) * 0.5;
    
    // Apply cloud coverage - higher threshold = less clouds
    float threshold = mix(0.8, 0.2, cloudCoverage);  // 0.8 (few clouds) to 0.2 (many clouds)
    float fadeRange = 0.3;
    cloudPattern = smoothstep(threshold, threshold - fadeRange, cloudPattern);
    
    // Soft edges
    float edge = smoothstep(0.0, 0.2, cloudPattern) * smoothstep(1.0, 0.6, cloudPattern);
    
    // Mix cloud color with sky color
    vec3 finalColor = mix(skyColor.rgb, cloudColor.rgb, cloudPattern);
    
    // Alpha for transparency
    float alpha = cloudPattern * 0.9;
    
    fragColor = vec4(finalColor, alpha);
}
